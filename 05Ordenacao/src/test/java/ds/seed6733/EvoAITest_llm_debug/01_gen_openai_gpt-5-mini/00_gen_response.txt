  /**
   * REQ-001: Verify ordering handles negative values and preserves other elements.
   * Given: an array with a negative value at index 3
   * When: the array is sorted with parameter 5
   * Then: the negative value should appear at index 0
   */
  @Test(timeout = 4000)
  public void test0()  throws Throwable  {
      // Arrange
      int[] intArray0 = new int[8];
      intArray0[3] = -3127;

      // Act
      Ordenacao.ordena(intArray0, 5);

      // Assert: multiple verifications with messages to aid debugging
      assertEquals("Array length should remain 8", 8, intArray0.length);
      assertEquals("First element should be the negative value after sort", -3127, intArray0[0]);
      assertArrayEquals("Sorted array mismatch: negative should be first",
              new int[] {-3127, 0, 0, 0, 0, 0, 0, 0}, intArray0);

      // Boundary check: empty array should remain unchanged (lightweight additional check)
      int[] empty = new int[0];
      Ordenacao.ordena(empty, 0);
      assertArrayEquals("Empty array should stay empty", new int[0], empty);
  }

  /**
   * REQ-002: Instantiation of Ordenacao and basic empty-array behavior
   * Given: a new Ordenacao instance
   * When: no operation is performed
   * Then: construction should succeed and basic empty-array sort should be safe
   */
  @Test(timeout = 4000)
  public void test1()  throws Throwable  {
      // Act: construct
      Ordenacao ordenacao0 = new Ordenacao();

      // Assert: verify construction and a minimal behavior to increase fault detection
      assertNotNull("Ordenacao instance should not be null", ordenacao0);

      // Verify basic behavior: sorting an empty array is safe
      int[] arr = new int[0];
      Ordenacao.ordena(arr, 0);
      assertArrayEquals("Empty array should remain empty after sort", new int[0], arr);

      // Quick behavioral check: simple two-element sort
      int[] a = new int[] {1, 0};
      Ordenacao.ordena(a, 1);
      assertEquals("First element should be 0 after sort", 0, a[0]);
  }
