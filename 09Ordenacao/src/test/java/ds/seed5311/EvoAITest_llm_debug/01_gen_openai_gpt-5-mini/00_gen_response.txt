package ds.seed5311;

import org.junit.Test;
import org.junit.Before;
import static org.junit.Assert.*;
import org.junit.runner.RunWith;
import org.junit.experimental.categories.Category;
import ds.Ordenacao;
import org.evosuite.runtime.EvoRunner;
import org.evosuite.runtime.EvoRunnerParameters;
import java.util.Arrays;

interface UnitTest {}

@RunWith(EvoRunner.class)
@EvoRunnerParameters(mockJVMNonDeterminism = true, useVFS = true, useVNET = true, resetStaticState = true, useJEE = true)
@Category(UnitTest.class)
public class EvoAITest extends Ordenacao_ESTest_scaffolding {

  private int[] baseArray;

  @Before
  public void setUp() {
    // common test data used in tests
    baseArray = new int[] {5,4,3,2,1};
  }

  /**
   * REQ-001: mergeSort_whenSortingFullRange_sortsAscending
   * Given-When-Then: Given an unsorted array, when mergeSort is applied to full range, then array is sorted ascending.
   */
  @Test(timeout = 4000)
  @Category(UnitTest.class)
  public void test0() throws Throwable {
    // Given
    int[] arr = Arrays.copyOf(baseArray, baseArray.length);
    // When
    Ordenacao.mergeSort(arr, 0, arr.length - 1);
    // Then: multiple specific assertions with messages
    assertEquals("array length should remain unchanged", 5, arr.length);
    assertArrayEquals("array should be sorted ascending", new int[] {1,2,3,4,5}, arr);
  }

  /**
   * REQ-002: mergeSort_invalidRange_throwsIllegalArgumentException
   * Given an Ordenacao instance, when mergeSort is called with an invalid range (low > high), then IllegalArgumentException is expected.
   */
  @Test(timeout = 4000)
  @Category(UnitTest.class)
  public void test1() throws Throwable {
    // preserve original behavior of constructing the object
    Ordenacao ordenacao0 = new Ordenacao();
    // When/Then: verify invalid index range handling
    try {
      Ordenacao.mergeSort(new int[] {1,2}, 2, 1);
      fail("Expected IllegalArgumentException for invalid range");
    } catch (IllegalArgumentException ex) {
      assertNotNull("expected IllegalArgumentException", ex);
    }
  }
}
