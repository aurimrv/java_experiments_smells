package ds.seed5831;

import org.junit.Test;
import static org.junit.Assert.*;
import static org.evosuite.runtime.EvoAssertions.*;
import ds.Ordenacao;
import org.evosuite.runtime.EvoRunner;
import org.evosuite.runtime.EvoRunnerParameters;
import org.junit.runner.RunWith;

@RunWith(EvoRunner.class)
@EvoRunnerParameters(mockJVMNonDeterminism = true, useVFS = true, useVNET = true, resetStaticState = true, useJEE = true)
public class EvoAITest extends Ordenacao_ESTest_scaffolding {

  /** REQ-001: Tests for Ordenacao.mergeSort behavior (happy and error paths) */
  @Test(timeout = 4000, expected = ArrayIndexOutOfBoundsException.class)
  public void test0() throws Throwable {
    // Given: an array of length 8
    int[] intArray0 = new int[8];
    intArray0[3] = -1;
    // When: mergeSort is called with an invalid end index
    Ordenacao.mergeSort(intArray0, 0, 938);
    // Then: an ArrayIndexOutOfBoundsException is expected (declared in annotation)
  }

  @Test(timeout = 4000)
  public void test1() throws Throwable {
    // Given: an Ordenacao instance and an unsorted array
    Ordenacao ordenacao0 = new Ordenacao();
    int[] arr = new int[] {3, 1, 2};
    // When: mergeSort is applied to the full array
    Ordenacao.mergeSort(arr, 0, arr.length - 1);
    // Then: array must be sorted and the instance must be non-null
    assertArrayEquals("Array should be sorted", new int[] {1, 2, 3}, arr);
    assertNotNull("Ordenacao instance should not be null", ordenacao0);
  }
}
